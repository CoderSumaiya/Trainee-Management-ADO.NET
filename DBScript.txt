CREATE TABLE [dbo].[Trainee]
(
	[TraineeId] INT NOT NULL PRIMARY KEY IDENTITY(1,1),
	TraineeName varchar(25) NOT NULL,
	MobileNo varchar(11) NOT NULL,
	IsEnrolled bit not null,
	AdmissionDate Datetime NOT NULL,
	ImageUrl varchar(250),
	CourseId int not null,
	RegFee int not null
)
CREATE TABLE [dbo].[Course]
(
	[CourseId] INT NOT NULL PRIMARY KEY IDENTITY(1,1),
	CourseName varchar(20) NOT NULL
)
CREATE TABLE [dbo].[CourseModule]
(
	[CourseModuleId] INT NOT NULL PRIMARY KEY IDENTITY(200,1),
	ModuleName varchar(20) NOT NULL,
	Duration int not null,
	TraineeId int not null REFERENCES Trainee(TraineeId),

)
